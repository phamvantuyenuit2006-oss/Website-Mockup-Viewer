4. TRUY VẤN ĐỆ QUY PHÂN CẤP QUẢN LÝ (RECURSIVE CTE)

Câu G1: Xuất toàn bộ cây phân cấp quản lý từ Giám đốc (Cấp 1) đến toàn thể nhân viên cấp dưới
WITH OrgChart_CTE AS ( -- Anchor Member: Ban Giam Doc (SupervisorID IS NULL) SELECT EmpID, FullName, SupervisorID, DeptID, 1 AS OrgLevel, CAST(FullName AS NVARCHAR(MAX)) AS ManagementPath FROM EMPLOYEE WHERE SupervisorID IS NULL UNION ALL -- Recursive Member: Nhan vien cap duoi SELECT e.EmpID, e.FullName, e.SupervisorID, e.DeptID, o.OrgLevel + 1, CAST(o.ManagementPath + ' -> ' + e.FullName AS NVARCHAR(MAX)) FROM EMPLOYEE e JOIN OrgChart_CTE o ON e.SupervisorID = o.EmpID ) SELECT EmpID, FullName, OrgLevel, ManagementPath FROM OrgChart_CTE ORDER BY OrgLevel, EmpID;
